{{- /* Markdown images in post bodies: resized, lazy-loaded, with intrinsic size to avoid layout shift. */ -}} {{- $src := .Destination -}} {{- $w := 0 -}}{{- $h := 0 -}} {{- if hasPrefix $src "/" -}} {{- $file := printf "static%s" $src -}} {{- if fileExists $file -}}{{- with imageConfig $file -}}{{- $w = .Width -}}{{- $h = .Height -}}{{- end -}}{{- end -}} {{- end -}} {{- /* */ -}}
